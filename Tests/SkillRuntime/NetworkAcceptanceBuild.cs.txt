using System;
using System.IO;
using System.Linq;
using SkillRuntime.Integration;
using UnityEditor;
using UnityEditor.Build;
using UnityEditor.Build.Reporting;
using UnityEngine;
using UnityEngine.SceneManagement;

// Staged into Assets/Editor only for this acceptance build. Never saves a scene.
public sealed class NetworkAcceptanceBuild : IProcessSceneWithReport
{
    static bool armed;
    public int callbackOrder => 10000;
    public void OnProcessScene(Scene scene, BuildReport report)
    {
        if (!armed || report == null || scene.path != "Assets/Scenes/BattleScene.unity") return;
        var manager = scene.GetRootGameObjects().SelectMany(o => o.GetComponentsInChildren<GameManager>(true)).Single();
        var settings = manager.GetComponent<PilotMatchSettings>() ?? manager.gameObject.AddComponent<PilotMatchSettings>();
        settings.mode = BattleAuthorityMode.DeterministicPilot;
        settings.profile = AssetDatabase.LoadAssetAtPath<PilotBattleProfile>(PilotArenaSetup.ProfilePath);
        if (!settings.profile) throw new BuildFailedException("Missing sealed pilot profile");
        Debug.Log("[NetworkAcceptanceBuild] Pilot added to BUILD COPY ONLY: " + scene.path);
    }
    public static void Run()
    {
        try
        {
            CheckDesktopReader();
            AssetDatabase.LoadAssetAtPath<PilotBattleProfile>(PilotArenaSetup.ProfilePath).ValidateAndFreeze();
            Directory.CreateDirectory("Builds/Phase4A-Network");
            armed = true;
            var report = BuildPipeline.BuildPlayer(new BuildPlayerOptions {
                scenes = EditorBuildSettings.scenes.Where(s => s.enabled).Select(s => s.path).ToArray(),
                locationPathName = "Builds/Phase4A-Network/PilotNetwork.exe",
                target = BuildTarget.StandaloneWindows64, options = BuildOptions.Development
            });
            armed = false;
            if (report.summary.result != BuildResult.Succeeded) throw new Exception(report.summary.result.ToString());
            Debug.Log("NETWORK_ACCEPTANCE_BUILD_COMPLETE bytes=" + report.summary.totalSize);
            EditorApplication.Exit(0);
        }
        catch (Exception ex) { Debug.LogException(ex); EditorApplication.Exit(1); }
    }
    static void Check(bool condition, string name)
    {
        if (!condition) throw new Exception("RESOURCE_FIX_FAIL " + name);
        Debug.Log("RESOURCE_FIX_PASS " + name);
    }
    static void CheckDesktopReader()
    {
        string path = Path.Combine(Path.GetFullPath("Library"), "shared-read-" + Guid.NewGuid().ToString("N") + ".txt");
        var go = new GameObject("Temporary desktop reader checks");
        try
        {
            var reader = go.AddComponent<ClassForUpdate>();
            string expected = File.ReadAllText(Path.Combine(Application.streamingAssetsPath, "Desktopspeed.txt"));
            File.WriteAllText(path, expected);
            int callbacks = 0;
            using (var otherClient = new FileStream(path, FileMode.Open, FileAccess.Read, FileShare.Read))
            {
                reader.GetFileStringFromStreamingAssets(path, s => { Check(s == expected, "Read succeeds while another shared reader holds the table"); callbacks++; });
            }
            Check(callbacks == 1, "Success callback invoked once");
            File.SetAttributes(path, FileAttributes.ReadOnly);
            reader.GetFileStringFromStreamingAssets(path, s => Check(s == expected, "Read-only file does not require write permission"));
            File.SetAttributes(path, FileAttributes.Normal);
            reader.GetFileStringFromStreamingAssets(path, s => {
                using (var exclusive = new FileStream(path, FileMode.Open, FileAccess.Read, FileShare.None))
                    Check(exclusive.Length > 0, "File is closed before parser callback");
            });
            bool missing = false;
            try { reader.GetFileStringFromStreamingAssets(path + ".missing", s => callbacks++); }
            catch (IOException ex) { missing = ex.Message.Contains(path) && ex.InnerException is FileNotFoundException; }
            Check(missing && callbacks == 1, "Missing file propagates path and cause without success callback");
            var parserError = new FormatException("Expected parser failure");
            bool parsing = false;
            try { reader.GetFileStringFromStreamingAssets(path, s => { throw parserError; }); }
            catch (InvalidDataException ex) { parsing = ex.Message.Contains(path) && ReferenceEquals(ex.InnerException, parserError); }
            Check(parsing, "Parser failure propagates path and original exception");
            using (var exclusive = new FileStream(path, FileMode.Open, FileAccess.ReadWrite, FileShare.None))
                Check(exclusive.Length > 0, "Parser failure leaves no file lock");
        }
        finally
        {
            UnityEngine.Object.DestroyImmediate(go);
            if (File.Exists(path)) { File.SetAttributes(path, FileAttributes.Normal); File.Delete(path); }
        }
    }
}
