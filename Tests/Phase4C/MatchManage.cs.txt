using System.Collections.Generic;

public struct MatchUserInfo
{
    public int uid;
    public int roleID;
    public MatchUserInfo(int uid, int roleID) { this.uid = uid; this.roleID = roleID; }
}

public class MatchManage
{
    private readonly List<MatchUserInfo> list_matchQueue = new List<MatchUserInfo>();
    // A popped queue member stays reserved until BattleService publishes currentBattleId.
    private readonly HashSet<int> pendingAssignment = new HashSet<int>();
    private static readonly object mmlockObj = new object();
    private static MatchManage instance;
    public static MatchManage Instance { get { lock (mmlockObj) return instance ?? (instance = new MatchManage()); } }
    private MatchManage() { }
    public void Creat() { }
    public void Destroy() { lock (mmlockObj) instance = null; }
    public void Assigned(int uid) { lock (mmlockObj) pendingAssignment.Remove(uid); }

    public void NewMatchUser(int uid, int roleId)
    {
        lock (mmlockObj)
        {
            if (UserManage.Instance.GetUserInfo(uid).matching || !pendingAssignment.Add(uid)) return;
            list_matchQueue.Add(new MatchUserInfo(uid, roleId));
            if (list_matchQueue.Count < ServerConfig.battleUserNum) return;
            var members = new List<MatchUserInfo>();
            for (int i = 0; i < ServerConfig.battleUserNum; i++)
            {
                members.Add(list_matchQueue[0]); list_matchQueue.RemoveAt(0);
            }
            ServerGlobal.Instance.AddAction(() => BattleManage.Instance.BeginBattle(members));
        }
    }
    public void CancleMatch(int uid)
    {
        lock (mmlockObj)
        {
            for (int i = 0; i < list_matchQueue.Count; i++)
                if (list_matchQueue[i].uid == uid)
                { list_matchQueue.RemoveAt(i); pendingAssignment.Remove(uid); break; }
        }
    }
    public void MatchSure(int matchID, int uid) { }
    public void CancleMatchSure(int matchID, int uid) { }
}
