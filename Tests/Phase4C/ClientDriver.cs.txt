#if DEVELOPMENT_BUILD || UNITY_EDITOR
using System;
using System.Collections;
using GameProtocol;
using SkillRuntime.Integration;
using UnityEngine;
using UnityEngine.SceneManagement;

// Actual login/match and outbound actions only. No synthetic incoming frames or HP writes.
[DefaultExecutionOrder(32000)]
public sealed class Phase4CClientDriver : MonoBehaviour
{
    string label, match; int completed; bool requested, attacked, confirmed; float started;
    UdpMessageDispatcher.DelegateReceiveMessage<UdpDownGameOver> receiver;
    bool dropped;
    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    static void Boot()
    {
        var args = Environment.GetCommandLineArgs(); int i = Array.IndexOf(args, "-phase4CClient");
        if (i < 0) return;
        var go = new GameObject("Phase4C acceptance outbound driver"); DontDestroyOnLoad(go);
        go.AddComponent<Phase4CClientDriver>().label = args[i + 1];
    }
    IEnumerator Start()
    {
        started = Time.realtimeSinceStartup; Application.runInBackground = true; Application.targetFrameRate = 60;
        QualitySettings.SetQualityLevel(0, true); QualitySettings.vSyncCount = 0;
        Application.SetStackTraceLogType(LogType.Log, StackTraceLogType.None);
        yield return new WaitForSecondsRealtime(2);
        LoginManager.Instance.OnClickLogin();
    }
    void LateUpdate()
    {
        if (Time.realtimeSinceStartup - started > 180) { Fail("timeout"); return; }
        if (SceneManager.GetActiveScene().name == "MainScene")
        {
            if (match != null)
            {
                if (!confirmed || PilotResultFlow.Active != null) { Fail("return without confirmation/retirement"); return; }
                completed++; Debug.Log("[Phase4CTest] RETURN client=" + label + " match=" + match + " completed=" + completed);
                match = null; requested = attacked = confirmed = false; receiver = null; dropped = false;
            }
            if (completed == 3) { Debug.Log("PHASE4C_CLIENT_COMPLETE client=" + label + " matches=3"); Application.Quit(0); enabled = false; return; }
            if (!requested)
            {
                requested = true;
                GameBlackboard.Instance.SetGameData<CharacterNameList>(GameConfig.ROLE_CURRENTNAME, CharacterNameList.Xingjianya);
                MatchManager.Instance.SendRequestMatchRequest();
            }
            return;
        }
        var roles = BattleManager.roleManage;
        if (!roles || !roles.initFinish || roles.Pilot == null) return;
        var pilot = roles.Pilot;
        if (pilot.Faulted) { Fail(pilot.Fault); return; }
        int slot = BattleData.Instance.battleID;
        if (match == null)
        {
            match = BattleData.Instance.MatchId;
            if (pilot.Health(1).ReadPilotHealth() != 20 || pilot.Health(2).ReadPilotHealth() != 20 || pilot.Runtime.LastCompletedFrame != 0 || pilot.Cast(1) != null || pilot.Cast(2) != null)
            { Fail("fresh match state invalid"); return; }
            Debug.Log("[Phase4CTest] JOIN client=" + label + " scenario=" + completed + " match=" + match + " slot=" + slot + " HP=20,20 F=0 digest=" + pilot.Digest);
        }
        // Seed reliable action before first Ready (0.5s). Server waits for both initial input snapshots.
        if (!attacked)
        {
            attacked = true;
            if (completed == 2 || slot == completed + 1) BattleData.Instance.UpdateRightOperation(RightOpType.rop1, slot == 1 ? 0 : 60, 0);
        }
        var flow = PilotResultFlow.Active;
        // Real incoming first business confirmation is lost for A. Original scoped retry must recover.
        if (flow != null && receiver == null && label == "A")
        {
            receiver = UdpMessageDispatcher.Instance.mes_down_game_over;
            UdpMessageDispatcher.Instance.mes_down_game_over = msg => {
                if (!dropped && msg.status == (int)MatchLifecycle.AgreementStatus.Confirmed)
                { dropped = true; Debug.Log("[Phase4CTest] DROP_FIRST_CONFIRM match=" + match); return; }
                receiver?.Invoke(msg);
            };
        }
        if (flow == null || flow.Agreement.Confirmed == null || confirmed) return;
        var result = flow.Agreement.Confirmed;
        if (result.Winner != (completed == 2 ? 0 : completed + 1) || result.Result != (completed == 2 ? MatchLifecycle.Outcome.Draw : MatchLifecycle.Outcome.Winner) || flow.Agreement.ApplyCount != 1)
        { Fail("unexpected final result"); return; }
        if (completed == 2 && (!pilot.Health(1).IsDead || !pilot.Health(2).IsDead)) { Fail("draw requires both committed deaths"); return; }
        confirmed = true;
        Debug.Log("[Phase4CTest] PASS client=" + label + " match=" + match + " F=" + result.Frame + " result=" + result.Result + " winner=" + result.Winner + " hash=" + result.Hash);
        StartCoroutine(ReturnLater(flow));
    }
    IEnumerator ReturnLater(PilotResultFlow flow)
    {
        // Different UI/death tail delays must not advance authority or change result.
        var result = flow.Agreement.Confirmed;
        yield return new WaitForSecondsRealtime(label == "A" ? 1.2f : .3f);
        if (flow.Agreement.Confirmed != result || BattleManager.roleManage.Pilot.Runtime.LastCompletedFrame != result.Frame) { Fail("terminal frame advanced"); yield break; }
        flow.RequestReturn(); flow.RequestReturn();
    }
    void Fail(string reason) { Debug.LogError("PHASE4C_CLIENT_FAIL client=" + label + " " + reason); enabled = false; Application.Quit(1); }
}
#endif
