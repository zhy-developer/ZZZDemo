using System;
using System.IO;
using System.Linq;
using SkillRuntime.Integration;
using UnityEngine;
using UnityEditor;
using UnityEditor.Build;
using UnityEditor.Build.Reporting;
using UnityEngine.SceneManagement;

// Staged only for acceptance. Explicit fixture: HP=20, Hurt radius=10 world units.
// Keeps real authored attacks, Players, consumers, resolver and TCP/UDP path.
public sealed class Phase5ClientBuild : IProcessSceneWithReport
{
    static bool armed;
    public int callbackOrder => 10000;
    public void OnProcessScene(Scene scene, BuildReport report)
    {
        if (!armed || report == null || scene.path != "Assets/Scenes/BattleScene.unity") return;
        var manager = scene.GetRootGameObjects().SelectMany(x => x.GetComponentsInChildren<GameManager>(true)).Single();
        var settings = manager.GetComponent<PilotMatchSettings>() ?? manager.gameObject.AddComponent<PilotMatchSettings>();
        settings.mode = BattleAuthorityMode.DeterministicPilot;
        settings.profile = UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<PilotBattleProfile>(PilotArenaSetup.ProfilePath));
        settings.profile.name = "Phase5 explicit terminal fixture";
        bool soak = Array.IndexOf(Environment.GetCommandLineArgs(), "-phase5Soak") >= 0; settings.profile.testHP = soak ? 100 : 20; settings.profile.testHurtRadius = soak ? 5000 : 100000;
        settings.profile.testParameterNotice = soak ? "PHASE5 DEVELOPMENT SOAK: original development HP100/Hurt5000; not approved formal gameplay." : "PHASE5 TERMINAL TEST FIXTURE ONLY: HP20/Hurt100000; not gameplay configuration.";
        settings.profile.sealedSourceHash = settings.profile.SourceHash();
        settings.profile.ValidateAndFreeze();
        Debug.Log("[Phase5Fixture] BUILD COPY ONLY HP="+settings.profile.testHP+" HurtRadius="+settings.profile.testHurtRadius);
    }
    public static void Run()
    {
        try
        {
            AssetDatabase.LoadAssetAtPath<PilotBattleProfile>(PilotArenaSetup.ProfilePath).ValidateAndFreeze();
            armed = true; Directory.CreateDirectory("Builds/Phase5-Network"); Directory.CreateDirectory("Builds/Phase5-Soak");
            var result = BuildPipeline.BuildPlayer(new BuildPlayerOptions {
                scenes = EditorBuildSettings.scenes.Where(x => x.enabled).Select(x => x.path).ToArray(),
                locationPathName = Array.IndexOf(Environment.GetCommandLineArgs(), "-phase5Soak") >= 0 ? "Builds/Phase5-Soak/PilotNetwork.exe" : "Builds/Phase5-Network/PilotNetwork.exe", target = BuildTarget.StandaloneWindows64, options = BuildOptions.Development });
            if (result.summary.result != BuildResult.Succeeded) throw new Exception(result.summary.result.ToString());
            Debug.Log("PHASE5_CLIENT_BUILD_COMPLETE"); EditorApplication.Exit(0);
        }
        catch (Exception e) { Debug.LogException(e); EditorApplication.Exit(1); }
    }
}
