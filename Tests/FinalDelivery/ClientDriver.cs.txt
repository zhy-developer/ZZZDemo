#if DEVELOPMENT_BUILD || UNITY_EDITOR
using System;
using System.Collections;
using System.Linq;
using SkillRuntime.Integration;
using UnityEngine;
using UnityEngine.InputSystem;
using UnityEngine.InputSystem.LowLevel;
using UnityEngine.SceneManagement;

// Drives the actual UI buttons and existing keyboard/mouse actions only. Never injects operations,
// changes a profile, HP, role transform, mode, received frames, server address or skill bindings.
[DefaultExecutionOrder(32000)]
public sealed class DeliveryClientDriver : MonoBehaviour
{
    string label, output, match, buffered;
    bool requested, confirmed, pressed;
    int completed, clicks; long lastClick = -100;
    float started, menuSince;
    Phase5FrameRecorder recorder;
    Keyboard keyboard; Mouse mouse; InputSettings original, local;
    static string Arg(string name) { var a=Environment.GetCommandLineArgs();int i=Array.IndexOf(a,name);return i>=0&&i+1<a.Length?a[i+1]:null; }
    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)] static void Boot() {
        if(Arg("-deliveryClient")==null)return;
        var go=new GameObject("Delivery normal UI/input acceptance");DontDestroyOnLoad(go);go.AddComponent<DeliveryClientDriver>();
    }
    IEnumerator Start() {
        label=Arg("-deliveryClient");output=Arg("-deliveryOutput");started=Time.realtimeSinceStartup;
        Application.runInBackground=true;QualitySettings.vSyncCount=0;Application.targetFrameRate=label=="A"?60:30;
        Application.SetStackTraceLogType(LogType.Log,StackTraceLogType.None);
        yield return new WaitForSecondsRealtime(2);
        var ui=FindObjectOfType<LoginUI>();
        if(!ui||!ui.loginBtn){Fail("LoginScene button missing");yield break;}
        Debug.Log("[DeliveryUI] login button server="+NetConfig.ServerIP);ui.loginBtn.onClick.Invoke();
    }
    void SetupInput() {
        if(keyboard!=null)return;
        original=InputSystem.settings;local=Instantiate(original);local.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;InputSystem.settings=local;
        keyboard=InputSystem.AddDevice<Keyboard>();mouse=InputSystem.AddDevice<Mouse>();
        CharacterInputSystem.Instance.inputActions.asset.devices=new InputDevice[]{keyboard,mouse};
    }
    void LateUpdate() {
        if(label==null)return;
        if(Time.realtimeSinceStartup-started>240){Fail("bounded run timeout");return;}
        if(SceneManager.GetActiveScene().name=="MainScene") {
            if(match!=null) {
                if(!confirmed||PilotResultFlow.Active!=null){Fail("return without retirement");return;}
                recorder.Dispose();recorder=null;completed++;
                Debug.Log("[Delivery] RETURN client="+label+" match="+match+" completed="+completed+
                    " audio="+Resources.FindObjectsOfTypeAll<AudioSource>().Length+" particles="+Resources.FindObjectsOfTypeAll<ParticleSystem>().Length);
                match=null;requested=confirmed=false;buffered=null;lastClick=-100;menuSince=Time.realtimeSinceStartup;
            }
            if(completed==2){Debug.Log("DELIVERY_CLIENT_COMPLETE client="+label+" matches=2 clicks="+clicks);enabled=false;Application.Quit(0);return;}
            if(!requested&&Time.realtimeSinceStartup-menuSince>1) {
                var ui=FindObjectOfType<MainUI>();if(!ui)return;
                requested=true;Debug.Log("[DeliveryUI] match button");ui.btn_match.onClick.Invoke();
            }
            return;
        }
        var roles=BattleManager.roleManage;if(!roles||!roles.initFinish)return;
        var p=roles.Pilot;if(p==null){Fail("normal BattleScene still Legacy");return;}
        if(p.Faulted){Fail(p.Fault);return;}
        int slot=BattleData.Instance.battleID;
        if(match==null) {
            var settings=roles.GetComponent<PilotMatchSettings>();
            if(!settings||!settings.profile.normalGameplay||p.Health(1).ReadPilotHealth()!=100||p.Runtime.LastCompletedFrame!=0){Fail("normal configuration/fresh state");return;}
            match=BattleData.Instance.MatchId;SetupInput();
            recorder=new Phase5FrameRecorder(output,match,true);p.Diagnostics=recorder;
            Debug.Log("[Delivery] JOIN client="+label+" match="+match+" slot="+slot+" digest="+p.Digest+" normalGameplay=true");
        }
        bool click=false; var keys=new System.Collections.Generic.List<Key>();
        long frame=p.Runtime.LastCompletedFrame;
        if(p.TerminalSnapshot==null&&slot==completed+1) {
            var role=roles.GetRoleFromBattleID(slot);var other=roles.GetRoleFromBattleID(3-slot);
            var cast=p.Cast(slot);
            if(cast==null&&BattleData.Instance.PendingActionCount==0) {
                var delta=new Vector3(other.LogicalPosition.x-role.LogicalPosition.x,0,other.LogicalPosition.y-role.LogicalPosition.y);
                if(delta.sqrMagnitude>14000f*14000f) {
                    var direction=Quaternion.Inverse(Quaternion.Euler(0,role.Player.camera.eulerAngles.y,0))*delta.normalized;
                    if(direction.x>.35f)keys.Add(Key.D);else if(direction.x<-.35f)keys.Add(Key.A);
                    if(direction.z>.35f)keys.Add(Key.W);else if(direction.z<-.35f)keys.Add(Key.S);
                } else if(frame-lastClick>4&&!pressed)click=true;
            } else if(cast!=null&&cast.Skill.SkillId!="miyabi-normal-5"&&buffered!=cast.Key.ToString()&&BattleData.Instance.PendingActionCount==0&&!pressed) {
                var gate=cast.Skill.Events.First(e=>e.EventId=="combo.preinput");
                if(cast.SkillFrame>=gate.Frame){click=true;buffered=cast.Key.ToString();}
            }
        }
        if(click) { keys.Add(Key.Space); keys.Add(Key.LeftShift); keys.Add(Key.E); keys.Add(Key.Q); }
        InputSystem.QueueStateEvent(keyboard,new KeyboardState(keys.ToArray()));
        InputSystem.QueueStateEvent(mouse,new MouseState().WithButton(MouseButton.Left,click));
        pressed=click;if(click){clicks++;lastClick=frame;Debug.Log("[DeliveryInput] click frame="+frame+" slot="+slot);}
        var flow=PilotResultFlow.Active;
        if(flow==null||flow.Agreement.Confirmed==null||confirmed)return;
        var result=flow.Agreement.Confirmed;
        if(flow.Agreement.ApplyCount!=1||result.Winner!=completed+1){Fail("unexpected terminal or repeat apply");return;}
        confirmed=true;Debug.Log("[Delivery] PASS client="+label+" match="+match+" F="+result.Frame+" result="+result.Result+" winner="+result.Winner+" hash="+result.Hash);
        StartCoroutine(Return(flow));
    }
    IEnumerator Return(PilotResultFlow flow) {yield return new WaitForSecondsRealtime(1);flow.RequestReturn();flow.RequestReturn();}
    void Fail(string reason){Debug.LogError("DELIVERY_CLIENT_FAIL "+reason);enabled=false;Application.Quit(1);}
    void OnDestroy(){recorder?.Dispose();if(keyboard!=null&&keyboard.added)InputSystem.RemoveDevice(keyboard);if(mouse!=null&&mouse.added)InputSystem.RemoveDevice(mouse);if(original)InputSystem.settings=original;if(local)Destroy(local);}
}
#endif
